## 结构体传参

```c
#include <stdio.h>

struct S {
    int data[1000];
    int num;
};

void print(struct S s) {
    for (int i = 0; i < 3; i++) {
        printf("%d\n", s.data[i]);
    }
    printf("%d\n", s.num);
}

void printp(struct S *s) {
    for (int i = 0; i < 3; i++) {
        printf("%d\n", s->data[i]);
    }
    printf("%d\n", s->num);
}

int main() {
    struct S s = { {1, 2, 3}, 100 };
    print(s);    // 传值调用
    printp(&s);  // 传址调用
    return 0;
}  // 更推荐传址调用
```
