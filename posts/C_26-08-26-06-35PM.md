## malloc 和 free

```c
#include <stdio.h>
#include <stdlib.h>

int main() {
    int arr[10] = {0};  // 栈

    int *p = (int *)malloc(40);  // 在堆里取40字节空间
    if (p == NULL) {
        return 1;
    }
    for (int i = 0; i < 10; i++) {
        *(p + i) = i;
    }
    for (int i = 0; i < 10; i++) {
        printf("%d ", *(p + i));
    }

    free(p);  // 释放40字节空间
    p = NULL;  // 清空指针保险

    return 0;
}
```
